% ========== SLIDE 2: PROBLEM =========================================
\begin{frame}{The Problem \& What We Built\speaker{Aysel Mamedova}}
  \small
  \begin{block}{Core idea}
    A user uploads a meal photo. The system identifies ingredients with portion sizes,
    looks up nutrition data from USDA, and returns \textbf{calories + macronutrients} in under 2 seconds.
  \end{block}

  \vspace{0.4em}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \textbf{\color{accent}Input}
      \begin{itemize}\setlength\itemsep{0.2em}
        \item Meal photo JPEG / PNG, \(\le\)5\,MB
        \item Validated at magic-byte level
      \end{itemize}

      \vspace{0.4em}
      \textbf{\color{accent}Entry Points}
      \begin{itemize}\setlength\itemsep{0.2em}
        \item \texttt{GET /} \quad Web UI (drag \& drop)
        \item \texttt{POST /analyze} \quad REST API
        \item \texttt{python -m foodanalyzer} \quad CLI
      \end{itemize}

    \column{0.50\textwidth}
      \textbf{\color{accent}Output per Ingredient}
      \begin{itemize}\setlength\itemsep{0.2em}
        \item Name + estimated grams
        \item kcal, protein, carbs, fat
        \item Confidence score from Gemini VLM
      \end{itemize}

      \vspace{0.4em}
      \textbf{\color{accent}Persistence}
      \begin{itemize}\setlength\itemsep{0.2em}
        \item Every analysis saved to PostgreSQL
        \item History: \texttt{GET /history?limit=N}
      \end{itemize}
  \end{columns}
\end{frame}

% ========== SLIDE 3: DESIGN DECISIONS ================================
\begin{frame}{Three Design Decisions We Can Defend\speaker{Aysel Mamedova}}
  \small
  \begin{enumerate}\setlength\itemsep{0.55em}
    \item \textbf{Composition over inheritance in \texttt{FoodAnalyzer}}
      \begin{itemize}
        \item \texttt{FoodAnalyzer} receives \texttt{AIService}, \texttt{NutritionPipeline}, \texttt{Repository} as constructor args
        \item \textit{Why:} it \emph{uses} them, not \emph{extends} them --- dependency injection makes unit tests trivial
      \end{itemize}

    \item \textbf{asyncio over threading for USDA fan-out}
      \begin{itemize}
        \item USDA lookups are I/O-bound (\(\approx\)500\,ms each) --- zero CPU work
        \item \textit{Rejected:} \texttt{ThreadPoolExecutor} (GIL contention, memory overhead)
      \end{itemize}

    \item \textbf{pydantic-settings for typed configuration}
      \begin{itemize}
        \item One \texttt{Settings} class reads \texttt{.env}, validates types, \textbf{fails fast} at startup
        \item \textit{Alternative rejected:} scattered \texttt{os.getenv()} that silently returns \texttt{None}
      \end{itemize}
  \end{enumerate}
\end{frame}

% ========== SLIDE 5: STORAGE & DATABASE ==============================
\begin{frame}{Storage \& Database\speaker{Rahima Karimova}}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \textbf{\color{accent}PostgreSQL via asyncpg}
      \begin{itemize}\setlength\itemsep{0.22em}
        \item Fully async --- no blocking threads
        \item Connection pool created at startup, closed at shutdown (FastAPI lifespan)
      \end{itemize}

      \vspace{0.4em}
      \textbf{\color{accent}Schema: \texttt{analyses} table}
      \begin{itemize}\setlength\itemsep{0.2em}
        \item \texttt{id} \quad UUID primary key
        \item \texttt{image\_path} \quad TEXT
        \item \texttt{result} \quad JSONB (full nutrition response)
        \item \texttt{created\_at} \quad TIMESTAMP, indexed \texttt{DESC}
      \end{itemize}

    \column{0.47\textwidth}
      \textbf{\color{accent}Why JSONB instead of a normalized schema?}
      \begin{itemize}\setlength\itemsep{0.22em}
        \item Requirements do not ask SQL filtering by ingredient name
        \item Single \texttt{SELECT}, no joins --- faster reads, simpler code
        \item Deliberate trade-off: acceptable for our use case
      \end{itemize}

      \vspace{0.4em}
      \textbf{\color{accent}Repository class}
      \begin{itemize}\setlength\itemsep{0.2em}
        \item \texttt{save(result)} --- writes one row
        \item \texttt{list\_recent(limit, offset)} --- paginated history
      \end{itemize}
  \end{columns}
\end{frame}

% ========== SLIDE 6: ORCHESTRATION & LOGGING =========================
\begin{frame}{Core Orchestration \& Structured Logging\speaker{Rahima Karimova}}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \textbf{\color{accent}FoodAnalyzer.analyze() --- 4 steps}
      \begin{enumerate}\setlength\itemsep{0.3em}
        \item Validate image bytes (magic-byte check)
        \item Call \texttt{AIService} \(\to\) Gemini VLM \emph{(slowest step)}
        \item Fan out: \texttt{NutritionPipeline.fetch\_all()} in parallel
        \item Save to PostgreSQL, return result
      \end{enumerate}

      \vspace{0.3em}
      \textit{Every entry point --- Web, API, CLI --- goes through this same 4-step flow.}

    \column{0.47\textwidth}
      \textbf{\color{accent}StructuredFormatter (logging\_config.py)}
      \begin{itemize}\setlength\itemsep{0.22em}
        \item Every log carries: \texttt{elapsed\_ms}, \texttt{ingredient\_count},
              \texttt{attempt}, \texttt{exc\_type}
        \item Default: human-readable text for development
        \item \texttt{LOG\_FORMAT=json} \(\to\) machine-readable JSON
      \end{itemize}

      \vspace{0.3em}
      \begin{block}{Why structured logging?}
        \small In production you search by field, not grep strings.
        Our logs support that from day one.
      \end{block}
  \end{columns}
\end{frame}

% ========== SLIDE 7: CONCURRENCY =====================================
\begin{frame}{Concurrency: the Benchmark\speaker{Gülnur Mammadova}}
  \small
  \textbf{How it works:}
  \texttt{NutritionPipeline.fetch\_all()} fans out \(N\) USDA lookups via
  \texttt{asyncio.gather}, bounded by \texttt{Semaphore(MAX\_PARALLEL=10)}.

  \vspace{0.5em}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \begin{center}\small
    \begin{tabular}{lccc}
      \toprule
      $N$ & \textbf{Sequential} & \textbf{Concurrent} & \textbf{Speedup} \\
      \midrule
      3 & 1.51\,s & 0.50\,s & 3.0\(\times\) \\
      5 & 2.50\,s & 0.53\,s & \textbf{4.8\(\times\)} \\
      7 & 3.51\,s & 0.52\,s & 6.7\(\times\) \\
      \bottomrule
    \end{tabular}
    \end{center}
    {\footnotesize Conditions: simulated 0.5\,s USDA RTT, caches cleared, Python 3.13.7, Windows 11}

    \column{0.47\textwidth}
    \textbf{New bottleneck after parallelisation:}
    \begin{itemize}\setlength\itemsep{0.25em}
      \item Gemini VLM call
      \item Single DB write at end
      \item Not the USDA fan-out anymore
    \end{itemize}

    \vspace{0.4em}
    \textbf{Why Semaphore = 10?}\\
    USDA free tier: 1,000 req/h. Semaphore keeps burst well within limit.
  \end{columns}
\end{frame}

% ========== SLIDE 8: ROBUSTNESS / AI SERVICE =========================
\begin{frame}{Robustness: AI Service \& Retry\speaker{Gülnur Mammadova}}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \begin{block}{Real failure we hit}
      \texttt{google-genai==0.3.0} does not support\\
      \texttt{Files.upload(file=...)}. Original code used a v1 API.
    \end{block}

    \vspace{0.4em}
    \textbf{What happened:}
    \begin{enumerate}\setlength\itemsep{0.18em}
      \item \texttt{tenacity} retried 4 times
      \item Logs showed \texttt{attempt=1..4, exc=TypeError}
      \item FastAPI returned HTTP 500 with clean JSON
    \end{enumerate}

    \vspace{0.3em}
    \textbf{Fix applied:}
    \begin{itemize}\setlength\itemsep{0.2em}
      \item Switched to \texttt{types.Part.from\_bytes()}
      \item Inline image bytes, no Files API
      \item Zero changes in SE layer
    \end{itemize}

    \column{0.47\textwidth}
    \textbf{AIService wrapping pattern:}
    \begin{itemize}\setlength\itemsep{0.22em}
      \item Retry: 4 attempts, 2--30\,s exponential backoff
      \item Timeout: 60\,s VLM, 20\,s USDA
      \item Structured logging at every attempt
      \item Response validated with Pydantic
    \end{itemize}

    \vspace{0.4em}
    \begin{block}{Key insight}
      \small Isolating \texttt{ai/} calls behind \texttt{AIService} meant the fix
      was one line in the adapter --- our SE layer was untouched.
    \end{block}
  \end{columns}
\end{frame}

% ========== SLIDE 9: NUTRITION CACHE & LIMITATIONS ===================
\begin{frame}{NutritionCache \& What We'd Do Next\speaker{Gülnur Mammadova}}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \textbf{\color{accent}How NutritionCache works}
      \begin{itemize}\setlength\itemsep{0.22em}
        \item Dict keyed on lowercased ingredient name
        \item TTL: 24\,h default (\texttt{NUTRITION\_CACHE\_TTL\_SECONDS})
        \item Lazy eviction --- stale entries removed on read
        \item \texttt{asyncio.Lock} prevents duplicate writes
      \end{itemize}

      \vspace{0.4em}
      \textbf{\color{accent}Where the system is brittle today}
      \begin{itemize}\setlength\itemsep{0.2em}
        \item Cache is \textbf{in-memory}: restart clears it
        \item No API authentication or rate limiting
        \item Regional dish names may miss USDA data
        \item CLI test coverage: \textbf{0\%}
      \end{itemize}

    \column{0.47\textwidth}
      \textbf{\color{accent}Given another week we would}
      \begin{enumerate}\setlength\itemsep{0.4em}
        \item Replace cache with \textbf{Redis} --- persistent, keyed by USDA food ID
        \item Add \textbf{GitHub Actions CI} pipeline
        \item Add \textbf{multi-provider failover}\\
              Gemini \(\to\) Anthropic on consecutive 5xx
      \end{enumerate}
  \end{columns}
\end{frame}
