% ========== SLIDE 2: PROBLEM =========================================
\begin{frame}{The Problem}
  \begin{block}{Core idea}
    A user uploads a meal photo. The system identifies ingredients with portion sizes,
    looks up nutrition data from USDA, and returns \textbf{calories + macronutrients} — all in under 2 seconds.
  \end{block}

  \vspace{0.6em}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \textbf{\color{accent}Input}
      \begin{itemize}\setlength\itemsep{0.25em}
        \item Meal photo (JPEG / PNG, \(\le\)5\,MB)
        \item Validated at magic-byte level — not just extension
      \end{itemize}
      \vspace{0.5em}
      \textbf{\color{accent}Entry Points}
      \begin{itemize}\setlength\itemsep{0.25em}
        \item \texttt{GET /} \quad Web UI (drag-and-drop)
        \item \texttt{POST /analyze} \quad REST API
        \item \texttt{python -m foodanalyzer analyze} \quad CLI
      \end{itemize}

    \column{0.50\textwidth}
      \textbf{\color{accent}Output per Ingredient}
      \begin{itemize}\setlength\itemsep{0.25em}
        \item Name + estimated grams
        \item kcal, protein, carbs, fat
        \item Confidence score from VLM
      \end{itemize}
      \vspace{0.4em}
      \textbf{\color{accent}Persistence}
      \begin{itemize}\setlength\itemsep{0.25em}
        \item Every analysis saved to PostgreSQL
        \item History: \texttt{GET /history?limit=N}
      \end{itemize}
  \end{columns}
\end{frame}

% ========== SLIDE 4: DESIGN DECISIONS ================================
\begin{frame}{Three Design Decisions We Can Defend}

  \begin{enumerate}\setlength\itemsep{0.8em}
    \item \textbf{Composition over inheritance in \texttt{FoodAnalyzer}}
      \begin{itemize}
        \item \texttt{FoodAnalyzer} receives \texttt{AIService} + \texttt{NutritionPipeline} as constructor args
        \item \textit{Why:} it \emph{uses} them, not specialises them — dependency injection makes unit tests trivial
      \end{itemize}

    \item \textbf{asyncio over threading for USDA fan-out}
      \begin{itemize}
        \item USDA lookups are I/O-bound (\(\sim\)500\,ms each) — asyncio has zero OS-thread overhead
        \item \textit{Rejected:} \texttt{ThreadPoolExecutor} (GIL contention), \texttt{ProcessPoolExecutor} (CPU-bound only)
      \end{itemize}

    \item \textbf{pydantic-settings for typed configuration}
      \begin{itemize}
        \item One \texttt{Settings} class reads \texttt{.env}, validates types, fails fast at startup
        \item \textit{Alternative rejected:} scattered \texttt{os.getenv()} — no types, no early validation
      \end{itemize}
  \end{enumerate}

\end{frame}

% ========== SLIDE 5: CONCURRENCY =====================================
\begin{frame}{Concurrency: the Benchmark}

  \textbf{How it works:}
  \texttt{NutritionPipeline.fetch\_all()} fans out \(N\) USDA lookups via
  \texttt{asyncio.gather}, bounded by \texttt{Semaphore(MAX\_PARALLEL=10)}.

  \vspace{0.6em}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \begin{center}\small
    \begin{tabular}{lcccr}
      \toprule
      $N$ & \textbf{Sequential} & \textbf{Concurrent} & \textbf{Speedup} \\
      \midrule
      3 & 1.51\,s & 0.50\,s & 3.0\(\times\) \\
      5 & 2.50\,s & 0.53\,s & \textbf{4.8\(\times\)} \\
      7 & 3.51\,s & 0.52\,s & 6.7\(\times\) \\
      \bottomrule
    \end{tabular}
    \end{center}
    {\footnotesize Conditions: simulated 0.5\,s USDA RTT, caches cleared,\\Python 3.13.7, Windows 11}

    \column{0.47\textwidth}
    \textbf{New bottleneck after parallelisation:}
    \begin{itemize}\setlength\itemsep{0.3em}
      \item Gemini VLM call — sequential by design (one image \(\to\) one response)
      \item Single DB write at end
      \item \emph{Not} the USDA fan-out anymore
    \end{itemize}
    \vspace{0.4em}
    \textbf{Why Semaphore = 10?}\\
    USDA free tier: 1,000 req/h. Semaphore keeps burst well within limit.
  \end{columns}

\end{frame}

% ========== SLIDE 6: ROBUSTNESS ======================================
\begin{frame}{Robustness: One Concrete Failure Mode}

  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \begin{block}{Failure injected}
      \texttt{google-genai==0.3.0} does not support\\
      \texttt{Files.upload(file=...)} — original code used a v1 API.
    \end{block}

    \vspace{0.5em}
    \textbf{What happened:}
    \begin{enumerate}\setlength\itemsep{0.2em}
      \item \texttt{tenacity} retried 4 times — each failed
      \item Logs showed \texttt{attempt=1..4, exc=TypeError}
      \item FastAPI returned HTTP 500 with clean JSON — no traceback exposed
    \end{enumerate}

    \column{0.47\textwidth}
    \textbf{Fix applied:}
    \begin{itemize}\setlength\itemsep{0.25em}
      \item Switched to \texttt{types.Part.from\_bytes()}\\(inline image bytes — no files API needed)
      \item Zero changes in SE layer — only the AI adapter
    \end{itemize}

    \vspace{0.5em}
    \textbf{General wrapping pattern:}\\
    Every \texttt{ai.*} call goes through:
    \begin{itemize}\setlength\itemsep{0.2em}
      \item Retry (4 attempts, 2--30\,s backoff)
      \item Timeout (60\,s VLM / 20\,s USDA)
      \item Structured logging (\texttt{INFO} / \texttt{WARNING})
      \item Response validation (Pydantic)
    \end{itemize}
  \end{columns}

\end{frame}

% ========== SLIDE 9: LIMITATIONS =====================================
\begin{frame}{Limitations \& What We'd Do Next}

  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \textbf{\color{accent}Where the system is brittle today}
    \begin{itemize}\setlength\itemsep{0.35em}
      \item \textbf{In-memory cache is per-process}\\
            Restart clears the nutrition cache — Redis would fix this
      \item \textbf{Local filesystem for images}\\
            Multi-replica deployment needs S3 / GCS
      \item \textbf{No API authentication}\\
            \texttt{POST /analyze} is open — production needs rate limiting + keys
      \item \textbf{USDA text-match quality}\\
            Regional dish names may return zero nutrition silently
      \item \textbf{CLI coverage: 0\%}\\
            Manual verification only — \texttt{CliRunner} harness would close the gap
    \end{itemize}

    \column{0.47\textwidth}
    \textbf{\color{accent}Given another week we would}
    \begin{enumerate}\setlength\itemsep{0.5em}
      \item Add \textbf{Redis-backed} persistent nutrition cache
      \item Add \textbf{GitHub Actions CI}\\
            (lint + test + Docker build on every PR)
      \item Add \textbf{multi-provider failover}\\
            Gemini \(\to\) Anthropic on consecutive 5xx
    \end{enumerate}
  \end{columns}

\end{frame}
